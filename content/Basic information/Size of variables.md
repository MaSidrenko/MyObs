# Количество занимаемых байтов
1. Char - 1 byte;
2. bool - 1 byte;
3. int - 4 byte;
4. short - 2 byte;
5. long - 4 byte;
6. long long - 8 byte;
7. float - 4 byte;
8. double - 8 byte;
9. long double - 8 byte;
# Max & Min Value
## Формула:
### Для знаковых типов:
{-2^(n - 1) U 2^(n - 1) - 1}; Где n - Количество бит занимаемых типом;
### Для беззнаковых типов:
{0 U 2^n - 1}; Где n - Количество бит занимаемых типом;
### Подсчет
#### Знаковые типы:
1. Char = {-2^(8 - 1) U 2^(8 - 1) - 1} = {-128 U 127};
2. Bool = {True (Все что не ноль) U False(0)};
3. int = {-2^(32 - 1) U 2^(32 - 1) - 1} = {- 2 147 483 648 U 2 147 483 647};
4. short = {-2^(16 - 1) U 2^(16 - 1) - 1} = {-32 768 U 32 767};
5. long  = {-2^(32 - 1) U 2^(32 - 1) - 1} = {- 2 147 483 648 U 2 147 483 647};
6. long long = {-2^(64 - 1) U 2^(64 - 1) - 1} = {-9 223 372 036 854 775 808 U 9 223 372 036 854 775 807};
7. float = {2^(32 - 1) U 2^(32 - 1) - 1} = {-2 147 483 648 U 2 147 483 647} (3.4E +/- 38(Количество знаков после запятой) С точностью до семи цифр после запятой);
8. double {2^(64 - 1) U 2^(64 - 1) - 1} = {-2 147 483 648 U 2 147 483 647} (1.7E +/- 308(Количество знаков после запятой) с точностью до пятнадцати цифр после запятой);
9. long double = {2^(64 - 1) U 2^(64 - 1) - 1} = {-2 147 483 648 U 2 147 483 647} (1.7E +/- 308(Количество знаков после запятой) 
c точностью до пятнадцати цифр после запятой);
#### Беззнаковые типы:
